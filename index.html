-- MedQuiz : récompenses des examens blancs -- À exécuter une seule fois dans Supabase > SQL Editor. -- Les points sont enregistrés dans quiz_history.score afin d'alimenter -- les classements général et hebdomadaire existants. create table if not exists public.exam_reward_claims ( id bigint generated by default as identity primary key, user_id uuid not null references auth.users(id) on delete cascade, exam_key text not null, points_awarded integer not null check (points_awarded > 0), percentage integer not null check (percentage between 0 and 100), created_at timestamptz not null default now(), constraint exam_reward_claims_user_exam_unique unique (user_id, exam_key) ); alter table public.exam_reward_claims enable row level security; create or replace function public.award_exam_white_points( p_exam_key text, p_correct integer, p_total integer ) returns jsonb language plpgsql security definer set search_path = public, auth as $$ declare v_user_id uuid := auth.uid(); v_pct integer; v_points integer; v_inserted integer; begin if v_user_id is null then raise exception 'authentication_required'; end if; if p_exam_key is null or length(trim(p_exam_key)) < 8 or length(p_exam_key) > 100 then raise exception 'invalid_exam_key'; end if; if p_total is null or p_total <= 0 or p_correct is null or p_correct < 0 or p_correct > p_total then raise exception 'invalid_exam_result'; end if; v_pct := round((p_correct::numeric / p_total::numeric) * 100)::integer; v_points := case when v_pct >= 90 then 100 when v_pct >= 80 then 75 when v_pct >= 70 then 50 when v_pct >= 60 then 30 when v_pct >= 50 then 15 else 5 end; insert into public.exam_reward_claims(user_id, exam_key, points_awarded, percentage) values (v_user_id, p_exam_key, v_points, v_pct) on conflict (user_id, exam_key) do nothing; get diagnostics v_inserted = row_count; if v_inserted = 0 then return jsonb_build_object('already_awarded', true, 'points_awarded', 0, 'percentage', v_pct); end if; -- On réutilise la table de points déjà lue par les classements MedQuiz. insert into public.quiz_history(user_id, score, nombre_questions) values (v_user_id, v_points, 1); return jsonb_build_object('already_awarded', false, 'points_awarded', v_points, 'percentage', v_pct); end; $$; revoke all on function public.award_exam_white_points(text, integer, integer) from public; grant execute on function public.award_exam_white_points(text, integer, integer) to authenticated;
